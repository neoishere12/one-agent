3. Update ServeHTTP to route /sse and /mcp:

func (s *Server) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    if s.handleHealth(w, r) {
        return
    }
    if s.handleIngest(w, r) {
        return
    }

    switch r.URL.Path {
    case "/sse":
        if r.Method == http.MethodOptions {
            writeNoContent(w)
            return
        }
        s.handleSSE(w, r)
    case "/mcp":
        if r.Method == http.MethodOptions {
            writeNoContent(w)
            return
        }
        if r.Method == http.MethodPost {
            s.handleMCPPost(w, r)
            return
        }
        // GET /mcp — info response (keep existing behaviour)
        writeJSON(w, http.StatusOK, map[string]any{
            "status": "ok",
            "transport": map[string]any{"type": "jsonrpc-http"},
        })
    default:
        writeJSON(w, http.StatusNotFound, map[string]string{"error": "not found"})
    }
}
